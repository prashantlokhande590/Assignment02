#Even or Odd Function Write a function is_even(n) that returns "Even" if the number is even, otherwise"Odd".

def is_even(n):
    if n % 2 == 0:
        return "Even"
    else:
        return "Odd"

def largest(a, b):

#4. Sum of Numbers Write a function sum_n(n) that returns the sum of numbers from 1 to n.

def sum_n(n):
    total = 0
    for i in range(1, n + 1):
        total += i
    return total

def table(n):

#6. Count Even Numbers Write a function that takes a list and returns how many even numbers it contains.Example:count_even([1, 2, 3, 4, 6])Output:3

def count_even(lst):
    count = 0
    for num in lst:
        if num % 2 == 0:
            count += 1
    return count

#9. Palindrome Checker
#● Write a function that returns True if a string is a palindrome.

def is_palindrome(text):
    
    clean_text = str(text).lower()
    return clean_text == clean_text[::-1]
    
#10. Count Vowels Write a function that counts the vowels in a string.

def count_vowels(string):
    vowels = "aeiou"
    count = 0
    for char in string.lower():
        if char in vowels:
            count += 1
    return count

#11. Second Largest Number Write a function that returns the second largest number in a list without using sort().

def second_largest(lst):
    if len(lst) < 2:
        return None  
    
    
    largest = second = float('-inf')
    
    for num in lst:
        if num > largest:
            second = largest
            largest = num
        elif num > second and num != largest:
            second = num
            
    return second if second != float('-inf') else None

#12. Guessing Game
#● Generate a random number between 1 and 10.
#● Keep asking the user to guess until they get it right.
#● Use a loop, conditionals, and functions.

import random

def guessing_game():
    target = random.randint(1, 10)
    print("I have chosen a number between 1 and 10. Can you guess it?")
    
    while True:
        try:
            guess = int(input("Enter your guess: "))
            if guess == target:
                print("Congratulations! You got it right.")
                break
            elif guess < target:
                print("Too low! Try again.")
            else:
                print("Too high! Try again.")
        except ValueError:
            print("Please enter a valid integer.")


#13 Write a function grade(marks) that:
#● Returns "A" for marks ≥ 90
#● Returns "B" for marks ≥ 80
#● Returns "C" for marks ≥ 70
#● Returns "D" for marks ≥ 60
#● Otherwise returns "F"

def grade(marks):
    if marks >= 90:
        return "A"
    elif marks >= 80:
        return "B"
    elif marks >= 70:
        return "C"
    elif marks >= 60:
        return "D"
    else:
        return "F"
